// AI 判讀（判讀異常時才呼叫）：Gemini 免費版 API
// 移植原「新進住民體檢」LINE Bot（app.py）的設定：
//   - 多把金鑰輪替：GEMINI_API_KEY、GEMINI_API_KEY_2～4，撞額度（429）自動換下一把
//   - 模型清單好模型優先（settings.ai_models，可用 GEMINI_MODELS 機密設定覆寫）
//   - temperature 0、輸出 JSON、逐頁加「【檔案第 N 頁】」標記
//   - 判讀規則：同義詞、嚴禁憑空生成（特別是胸部X光）、定量 B 肝依參考界線判正負、梅毒 RPR／TPPA、
//     多份報告只取最新、數值逐字抄錄
// 金鑰只放在 Edge Function 機密設定，前端與資料庫都拿不到。
// 單一檔案（不引用 _shared），方便直接在 Supabase 後台的編輯器部署
import { createClient } from "npm:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": Deno.env.get("APP_ORIGIN") ?? "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};
function json(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" } });
}
function adminClient() {
  return createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
    { db: { schema: "health_exam" }, auth: { persistSession: false } });
}

const DEFAULT_MODELS = ["gemini-2.5-flash", "gemini-3.5-flash-lite", "gemini-2.5-flash-lite", "gemini-2.0-flash"];

interface PageIn { no: number; data: string; mime?: string }
interface ItemRow { id: string; code: string; name_zh: string; name_en: string | null; value_type: string; unit: string | null }

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: corsHeaders });
  const t0 = Date.now();

  // 只有系統內的帳號可以呼叫
  const caller = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_ANON_KEY")!, {
    db: { schema: "health_exam" },
    global: { headers: { Authorization: req.headers.get("Authorization") ?? "" } },
  });
  const { data: accountId } = await caller.rpc("current_account_id");
  if (!accountId) return json({ error: "請先登入系統" }, 401);

  const body = await req.json().catch(() => null) as
    { case_id?: string; reason?: string; pages?: PageIn[]; only?: string[] } | null;
  const pages = (body?.pages ?? []).filter((p) => p?.data).slice(0, 12);
  if (!pages.length) return json({ error: "沒有要判讀的頁面" }, 400);

  const keys = ["GEMINI_API_KEY", "GEMINI_API_KEY_2", "GEMINI_API_KEY_3", "GEMINI_API_KEY_4"]
    .map((k) => Deno.env.get(k)).filter((k): k is string => !!k);
  if (!keys.length) return json({ error: "尚未設定 Gemini 金鑰（GEMINI_API_KEY），請系統管理員到 Supabase 機密設定加入" }, 503);

  const admin = adminClient();
  const [{ data: items }, { data: aliases }, { data: setting }] = await Promise.all([
    admin.from("test_items").select("id,code,name_zh,name_en,value_type,unit").eq("active", true).order("sort_order"),
    admin.from("item_aliases").select("item_id,alias"),
    admin.from("settings").select("key,value").in("key", ["ai_models", "ai_enabled"]),
  ]);
  const envModels = Deno.env.get("GEMINI_MODELS")?.split(",").map((s) => s.trim()).filter(Boolean);
  const cfgModels = setting?.find((s) => s.key === "ai_models")?.value as string[] | undefined;
  const models = envModels?.length ? envModels : cfgModels?.length ? cfgModels : DEFAULT_MODELS;

  const only = new Set(body?.only ?? []);
  const list = (items as ItemRow[] ?? []).filter((i) => !only.size || only.has(i.code));
  const aliasOf = (id: string) => (aliases ?? []).filter((a) => a.item_id === id).map((a) => a.alias);
  const itemText = list.map((i) =>
    `${i.code}｜${i.name_zh}${i.name_en ? "｜" + i.name_en : ""}｜同義：${aliasOf(i.id).join("、") || "無"}｜類型：${TYPE_TEXT[i.value_type] ?? i.value_type}${i.unit ? "｜單位：" + i.unit : ""}`,
  ).join("\n");

  const parts: unknown[] = [];
  for (const p of pages) {
    parts.push({ text: `【檔案第 ${p.no} 頁】` });
    parts.push({ inline_data: { mime_type: p.mime ?? "image/jpeg", data: p.data } });
  }
  parts.push({ text: prompt(itemText) });

  let lastErr = "";
  for (const model of models) {
    for (let k = 0; k < keys.length; k++) {
      try {
        const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`, {
          method: "POST",
          headers: { "Content-Type": "application/json", "x-goog-api-key": keys[k] },
          body: JSON.stringify({
            contents: [{ role: "user", parts }],
            generationConfig: { temperature: 0, responseMimeType: "application/json" },
          }),
        });
        if (!res.ok) {
          lastErr = `${model}＋金鑰#${k + 1}：HTTP ${res.status} ${(await res.text()).slice(0, 200)}`;
          continue;   // 429 額度、404 模型不存在、5xx 都換下一個組合
        }
        const out = await res.json();
        const text = (out?.candidates?.[0]?.content?.parts ?? []).map((x: { text?: string }) => x.text ?? "").join("");
        const parsed = parseJson(text);
        if (!parsed) { lastErr = `${model}：回傳不是 JSON`; continue; }
        const result = clean(parsed, new Set(list.map((i) => i.code)), new Set(pages.map((p) => p.no)));
        await admin.from("ai_calls").insert({ account_id: accountId, case_id: body?.case_id ?? null, page_count: pages.length,
          reason: body?.reason ?? null, model, key_index: k + 1, status: "ok", item_count: result.items.length, ms: Date.now() - t0 });
        return json({ ...result, model, key_index: k + 1 });
      } catch (e) {
        lastErr = `${model}＋金鑰#${k + 1}：${String(e).slice(0, 200)}`;
      }
    }
  }
  await admin.from("ai_calls").insert({ account_id: accountId, case_id: body?.case_id ?? null, page_count: pages.length,
    reason: body?.reason ?? null, status: "failed", error: lastErr.slice(0, 500), ms: Date.now() - t0 });
  return json({ error: "AI 判讀失敗（可能是免費額度已用完，請稍後再試或手動輸入）", detail: lastErr }, 502);
});

const TYPE_TEXT: Record<string, string> = { numeric: "數值", posneg: "陰陽性", semiquant: "半定量", text: "文字" };

function parseJson(text: string): Record<string, unknown> | null {
  try { return JSON.parse(text); } catch { /* 有時包在 ```json 內 */ }
  const m = text.match(/\{[\s\S]*\}/);
  try { return m ? JSON.parse(m[0]) : null; } catch { return null; }
}

// 只留系統認得的項目代碼與頁次；座標轉成 0～1
function clean(raw: Record<string, unknown>, codes: Set<string>, pageNos: Set<number>) {
  const items: { code: string; value: string; page: number | null; bbox: [number, number, number, number] | null; sampled_on: string | null }[] = [];
  for (const it of (Array.isArray(raw.items) ? raw.items : []) as Record<string, unknown>[]) {
    const code = String(it.code ?? "").trim().toUpperCase();
    const value = String(it.value ?? "").trim();
    if (!codes.has(code) || !value || items.some((x) => x.code === code)) continue;
    const page = Number(it.page);
    const b = Array.isArray(it.box_2d) && it.box_2d.length === 4 ? (it.box_2d as number[]).map(Number) : null;
    // Gemini 座標慣例：[ymin, xmin, ymax, xmax]，0～1000
    const bbox = b && b.every((n) => Number.isFinite(n))
      ? [b[1] / 1000, b[0] / 1000, b[3] / 1000, b[2] / 1000] as [number, number, number, number] : null;
    items.push({ code, value, page: pageNos.has(page) ? page : null, bbox, sampled_on: it.sampled_on ? String(it.sampled_on) : null });
  }
  const names = Array.isArray(raw.subjects) ? (raw.subjects as unknown[]).map(String).filter(Boolean) : [];
  return { items, sampled_on: raw.sampled_on ? String(raw.sampled_on) : null, subjects: names };
}

function prompt(itemText: string): string {
  return `你是一個高度嚴謹的醫療數據審核員。請判讀上面的體檢報告影像，只輸出 JSON。
【項目清單】（只能使用這些代碼；左邊是代碼，後面是名稱與同義寫法）
${itemText}

【最高強制規則】
1. 【嚴禁憑空生成】只能填報告中實際看得到的內容。報告沒有做的項目一律不要輸出，嚴禁依常識、慣例或其他項目推測補齊。
   特別是胸部X光（CXR）：只有文件中確實出現胸部X光／CXR／胸部攝影的判讀文字時才填寫；沒有就不要輸出，嚴禁自行填入「無活動性肺部病灶」「心肺無異常」等常見結論。
2. 【同義詞】各檢驗所寫法不同（中文、英文縮寫、全名混用）。報告出現清單中任一寫法，就視為該項目有做，必須輸出。填表前請逐項自問：這個項目在報告裡有沒有用別的名字出現？
3. 【數值抄錄零容錯】逐字元抄錄報告印的數字，不可四捨五入、換算單位、補零或去零，不可把參考值當成結果；不要附上箭頭或 H／L 旗標。
   易混淆字元特別注意：0/8/6、1/7、2/Z、3/8、4/9、5/6，模糊時用參考範圍交叉驗證。無法確定的數值不要輸出（會由人工補）。
4. 【各類型輸出格式】
   - 數值：只填數字，例如 14、5.04。
   - 範圍：照抄，例如 2-3、1-3。
   - 半定量（尿糖、尿蛋白、潛血等）：(-)、(±)、(+)、(++)、(+++)、(++++)。「4+」寫成 (++++)。
   - 陰陽性：陰性寫 (-)、陽性寫 (+)；報告寫 Non-Reactive／Reactive 時照寫。
   - 定量抗原／抗體（例如 HBsAg 0.031、Anti-HBs 1.500，附參考範圍如 <1.0(-)）：這就是檢驗結果，不可當成未檢驗。
     依報告參考範圍判斷正負號，寫成「(-)0.031」「(+)1.500」。數值 < 陰性上限為 (-)，≥ 陽性下限為 (+)。
   - 梅毒：篩檢（STS／RPR／VDRL）與確認試驗（TPPA）分開判讀；篩檢陽性但 TPPA 陰性時填「(+)（TPPA複查陰性）」；篩檢陰性就填 (-) 或 Non-Reactive。
   - 文字（胸部X光等）：所有外語一律翻譯成繁體中文（台灣用語）。手寫判讀請仔細辨認後完整寫出，例如「心臟肥大、主動脈弓鈣化；左上肺尖疑有腫瘤，宜進一步檢查」。
5. 【日期】sampled_on 填一般項目（血液、尿液、生化）的採檢日或檢驗日，格式民國 YYY/MM/DD（例 115/08/24）。
   糞便三項（寄生蟲、阿米巴、桿菌性痢疾）若有自己的採檢日，填在該項目的 sampled_on。
   手寫日期年份容易誤判，請用報告其他印刷日期交叉確認，嚴禁把近期日期判成去年。
6. 【多份報告合併】若文件含多個不同檢驗日期的報告，每個項目只採用最新一次的結果，舊結果當作不存在。
7. 【頁碼】文件前面有「【檔案第 N 頁】」標記，page 填你採用數值的那一頁 N，不要用報告自己印的頁碼。
8. 【座標】box_2d 填該結果值在那一頁影像中的位置 [ymin, xmin, ymax, xmax]（0～1000）；只框結果值本身，不要框項目名稱。
9. 【受檢者】subjects 列出受檢者本人姓名；排除檢驗人員、醫師、審核者、簽章者。

輸出格式：
{"subjects": ["姓名"], "sampled_on": "115/08/24",
 "items": [{"code": "GOT", "value": "14", "page": 1, "box_2d": [120, 520, 140, 560], "sampled_on": "115/08/24"}]}`;
}
